		markings,

	\tikzset{
		-g>-/.style 2 args={
			compedge,
			draw=#1,
			decoration={
				markings,
				mark=at position 0.55 with {\arrow{Triangle[scale=1.5, fill=#2]}}
			}
		},
		-g>-/.default={MyText}{MyGold} % Default to black line, white fill
	}
	\tikzset{
		-g>>-/.style 2 args={
			compedge,
			draw=#1,
			decoration={
				markings,
				mark=at position 0.50 with {\arrow{Triangle[scale=1.5, fill=#2]}},
				mark=at position 0.60 with {\arrow{Triangle[scale=1.5, fill=#2]}}
			}
		},
		-g>>-/.default={MyText}{MyGold},
	}
	\tikzset{
		-g>>>-/.style 2 args={
			compedge,
			draw=#1,
			decoration={
				markings,
				mark=at position 0.5 with {\arrow{Triangle[scale=1.5, fill=#2]}},
				mark=at position 0.55 with {\arrow{Triangle[scale=1.5, fill=#2]}},
				mark=at position 0.6 with {\arrow{Triangle[scale=1.5, fill=#2]}}
			}
		},
		-g>>>-/.default={MyText}{MyGold}
	}

	\tikzstyle{walk}=[
	decoration={
		markings,
		mark=between positions 0 and 1 step 3pt with {
			\node[fill=MyText,
			signal from=west,
			signal to=east,
			signal,
			transform shape,
			minimum width=1pt,
			minimum height=3pt,
			inner sep=0]{};}},
	decorate]
